\newcommand{\person}[3]{
  \foreach \y / \r / \t in {
     2  / 180 / {\large #2},     -2  / 0 / {\large #2},
    3.9 / 180 / {\small\it Стол №\:#3}, -3.9 / 0 / {\small\it Стол №\:#3}
  } {
    \draw (#1) ++(0, \y) node[rotate=\r]{\begin{minipage}[c]{9.8cm}
      \centering \t%
    \end{minipage}};
  }
}

\newcommand{\tablepage}[8]{
  \newpage\tikz{
    \draw (-10.4, 0) -- (10.4, 0) (0, 14.7) -- (0, -14.7);
    \draw[dashed,black!45] \foreach \y in {12.5, 7.35, 2.2, -2.2, -7.35, -12.5}
      {(-10.4, \y) -- (10.4, \y)};
    \person{-5.2, 7.35}{#1}{#2}  \person{5.2, 7.35}{#3}{#4}
    \person{-5.2, -7.35}{#5}{#6}  \person{5.2, -7.35}{#7}{#8}
  }%
}
